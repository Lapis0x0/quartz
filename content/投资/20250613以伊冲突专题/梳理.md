以伊冲突速览：

<iframe src="以伊冲突速览.html" id="report-iframe" width="100%" scrolling="no" style="border:none; overflow:hidden;" onload="resizeIframe(this)"></iframe>

<script>
function resizeIframe(iframe) {
  iframe.style.height = iframe.contentWindow.document.body.scrollHeight + 10 + 'px';
}

window.addEventListener('resize', function() {
  var iframe = document.getElementById('report-iframe');
  if (iframe) {
    resizeIframe(iframe);
  }
});
</script>
